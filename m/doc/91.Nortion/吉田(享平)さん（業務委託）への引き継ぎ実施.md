            ```bash
            export DB_USERNAME=postgres
            export DB_PASSWORD=
            
            # cloud-sql-proxyをバックグラウンドで起動しておく
            # cloud-sql-proxyについては以下のURLを参照して取得
            # https://docs.cloud.google.com/sql/docs/mysql/sql-proxy?hl=ja
            ../cloud-sql-proxy medup-foro:asia-northeast1:medup-foro-stg --port 13307 &
            
            proxy_pid=$!
            
            # ssh接続完了まで適当な時間待つ
            sleep 2
            
            PGPASSWORD=$DB_PASSWORD psql -h localhost -p 13307 -U postgres -d foro_stg
            # PGPASSWORD=$DB_PASSWORD psql -h localhost -p 13307 -U postgres -d foro_stg -f ../20240807_segment_tests.sql -A -F, > ../20240807_segment_test.csv
            
            # cloud-sql-proxyを終了する
            kill $proxy_pid
            wait $proxy_pid
            ```

            ```bash
            #!/bin/bash
            ALLOYVM_USER=chihiro.naganuma
            ALLOYVM_SSHKEY=~/.ssh/foro-alloydb-vm
            IAM_USER=chihiro.naganuma@medup.jp
            
            # 事前にmedup-foroプロジェクトのalloy-vmにssh鍵を登録のこと
            # sshトンネルをバックグラウンドで起動しておく

            # ssh接続完了まで適当な時間待つ

            # sshトンネルを終了する
            kill $ssh_pid
            wait $ssh_pid
            ```
